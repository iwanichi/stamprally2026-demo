<body>

  <h1>デジタルスタンプラリー</h1>
  <p>各会場のボタンを押してスタンプを集めよう！</p>

  <div id="complete-msg">🎉 コンプリートおめでとう！</div>

  <div class="card-container" id="stampGrid"></div>

  <button class="reset-btn" onclick="resetStamps()">リセットする</button>

  <script>
    // スポット一覧の設定（名称を変更）
    const spots = [
      { id: 'spot1', name: '商工祭' },
      { id: 'spot2', name: '農業祭' },
      { id: 'spot3', name: '福の市' },
      { id: 'spot4', name: '住宅際' }
    ];

    // 保存データの取得
    let collected = JSON.parse(localStorage.getItem('stampData')) || {};

    function render() {
      const grid = document.getElementById('stampGrid');
      grid.innerHTML = '';
      let count = 0;

      spots.forEach(spot => {
        const isStamped = !!collected[spot.id];
        if (isStamped) count++;

        const card = document.createElement('div');
        card.className = `stamp-spot ${isStamped ? 'active' : ''}`;
        card.innerHTML = `
          <strong>${spot.name}</strong>
          <span class="stamp-mark">済</span>
          ${!isStamped ? `<button onclick="getStamp('${spot.id}')">スタンプ獲得</button>` : ''}
        `;
        grid.appendChild(card);
      });

      if (count === spots.length) {
        document.getElementById('complete-msg').style.display = 'block';
      } else {
        document.getElementById('complete-msg').style.display = 'none';
      }
    }

    // スタンプを押す処理
    function getStamp(id) {
      collected[id] = true;
      localStorage.setItem('stampData', JSON.stringify(collected));
      render();
    }

    // リセット処理
    function resetStamps() {
      if (confirm('スタンプを初期化しますか？')) {
        collected = {};
        localStorage.removeItem('stampData');
        render();
      }
    }

    // 初期描画
    render();
  </script>
